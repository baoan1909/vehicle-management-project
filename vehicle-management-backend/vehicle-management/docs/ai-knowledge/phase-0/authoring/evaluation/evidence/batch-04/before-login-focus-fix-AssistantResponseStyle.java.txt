    public static final String CONTEXTUAL_VERSION = "contextual-response-v7";

            return focus + "\nNgười dùng hỏi nơi mở chức năng tìm bãi. Chỉ vị trí màn hình và cách mở theo bằng chứng; không hỏi lại lỗi khi họ chưa báo lỗi.";

                + "Không liệt kê lại cả quy trình hoặc tự đoán bước tiếp theo. Nếu đang nói về tạo phiếu, hỏi họ dùng biểu mẫu hay thẻ yêu cầu trong chat; "

    private static final Pattern TECHNICAL_QUESTION = Pattern.compile(
            "\\b(?:api|enum|json|class|sql|backend|frontend|type|lap trinh|ma nguon)\\b");
    private static final Pattern INTERNAL_DETAIL = Pattern.compile(
            "\\b(?:PENDING(?:_PAYMENT|_CARD|_SPACE)?|ACTIVE|INACTIVE|BLOCKED|EXPIRED|TERMINATED|"
                    + "CANCELLED|REJECTED|RESOLVED|OPEN|CLOSED|LOST_CARD|CUSTOMER_DIRECT|ASSISTANT_SUPPORT|"
                    + "SUPPORT_REQUESTER_DIRECT|READ_OWN|READ_ALL|[A-Z]+(?:_[A-Z]+)+|"
                    + "subscriptionId|customerId|customerVehicleId|parkingSessionId|cardUid|qrAccessToken|(?i:widget|subscription)|OWN|ALL)\\b"
                    + "|/api/|\\b[0-9a-fA-F]{8}(?:-[0-9a-fA-F]{4}){3}-[0-9a-fA-F]{12}\\b");
    private static final Pattern ORDERED_STEP = Pattern.compile("(?<!\\S)([1-9])[.)][ \\t]+\\p{L}");
    private static final Pattern INLINE_STEP = Pattern.compile("(?<=\\S)[ \\t]+(?=[1-9][.)][ \\t]+\\p{L})");

    private AssistantResponseStyle() { }

    /** Whitespace-only presentation: never renumber steps or rewrite dates, amounts or instructions. */
    public static String formatForDisplay(String answer) {
        if (answer == null) return null;
        var steps = ORDERED_STEP.matcher(answer);
        int expected = 1;
        while (steps.find()) {
            if (Integer.parseInt(steps.group(1)) != expected++) return answer;
        }
        return expected > 2 ? INLINE_STEP.matcher(answer).replaceAll("\n") : answer;
    }

    public static boolean containsUnnecessaryInternalDetail(String answer, String question) {
        if (answer == null) return false;
        boolean technical = question != null
                && TECHNICAL_QUESTION.matcher(AssistantIntentRouter.matching(question)).find();
        return !technical && INTERNAL_DETAIL.matcher(answer).find();
    }
}
