    private AssistantQuestionResolver() { }

    public record ResolvedQuestion(String query, AssistantIntentRouter.IntentDecision decision, boolean contextual) { }

    /** Callers must load actor-filtered history for dependent knowledge questions as well as vague ones. */
    public static boolean requiresContext(String question) {
        var intent = AssistantIntentRouter.route(question).intent();
        return intent == AssistantIntent.CLARIFICATION
                || intent == AssistantIntent.STATIC_KNOWLEDGE && isDependentFollowUp(question);
    }

    /** History is newest first and must already be restricted to the actor and conversation by the caller. */
    public static ResolvedQuestion resolve(String question, List<String> previousQuestions) {
        var decision = AssistantIntentRouter.route(question);
        boolean needsTopic = requiresContext(question);
        if (needsTopic && previousQuestions != null) {
            for (String previous : previousQuestions.stream().limit(3).toList()) {
                var previousIntent = AssistantIntentRouter.route(previous).intent();
                if (previousIntent == AssistantIntent.CLARIFICATION) continue;
                if (previousIntent != AssistantIntent.STATIC_KNOWLEDGE) break;
                String focusedTopic = AssistantGuideFocus.from(previous).publicTopic();
                if (!focusedTopic.isBlank()) {
                    String contextual = "Hướng dẫn " + focusedTopic + ". " + question;
                    return new ResolvedQuestion(contextual, AssistantIntentRouter.route(contextual), true);
                }
                String folded = AssistantIntentRouter.matching(previous);
                for (int i = 0; i < TOPICS.size(); i++) {
                    if (!folded.matches(".*\\b" + TOPICS.get(i) + "\\b.*")) continue;
                    String contextual = "Hướng dẫn " + LABELS.get(i) + ". " + question;
                    return new ResolvedQuestion(contextual, AssistantIntentRouter.route(contextual), true);
                }
                // Do not search older topics when the latest eligible turn changed the subject.
                break;
            }
        }
        return new ResolvedQuestion(question, decision, false);
    }

    private static boolean isDependentFollowUp(String question) {
        String folded = AssistantIntentRouter.matching(question);
        if (!folded.matches("^(?:vay|the|neu|luc|con|ban dem|so do|dia chi do|nut do)\\b.*")) return false;
        if (folded.matches(".*\\b(?:tai khoan|mat khau|google|email)\\b.*")) return false;
        // A broad word such as registration cannot distinguish signing up from entering
        // without a parking subscription. Explicit new topics still start a fresh question.
        var broadTopics = java.util.Set.of("dang ky");
        return TOPICS.stream().filter(topic -> !broadTopics.contains(topic))
                .noneMatch(topic -> folded.matches(".*\\b" + topic + "\\b.*"));
    }
